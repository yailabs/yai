"""Replay the actual Markdown operator commands; this is not human acceptance."""
import json
import os
from pathlib import Path
import re
import subprocess
import time

root = Path('/home/mothx/computer-science/projects/YAI/yai')
document = root / 'refoundation/foundation-recovery/interlock-05/MANUAL-ACCEPTANCE.md'
blocks = re.findall(r'```sh\n(.*?)```', document.read_text(), re.S)
peers = []
shell = subprocess.Popen(['bash', '--noprofile', '--norc'], cwd=root,
                         stdin=subprocess.PIPE, stdout=subprocess.PIPE,
                         stderr=subprocess.STDOUT, text=True,
                         env=dict(os.environ, CARGO_NET_OFFLINE='true'))
ids = {}
plans = []
turns = []
realizations = []
order = 0
run_id = 'i05-manual-replay-' + str(os.getpid())
try:
    for block in blocks:
        for command in block.strip().splitlines():
            order += 1
            if command.startswith('python3 tests/fixtures/'):
                peer = subprocess.Popen(command.split(), cwd=root, stdout=subprocess.PIPE,
                                        stderr=subprocess.PIPE, text=True)
                peers.append(peer)
                port = peer.stdout.readline().strip()
                assert port in ('18351', '18352'), (port, command)
                print(json.dumps(dict(run_id=run_id, order=order, cwd=str(root), command=command,
                                      posture='fixture_running', stdout=port)), flush=True)
                continue
            submitted = None
            if command.startswith('read -r '):
                variable = command.split()[-1]
                key = ('target_id' if 'TARGET' in variable else
                       'evidence_id' if 'EVIDENCE' in variable else 'turn_id')
                submitted = ids[key]
            marker = f'__I05_COMMAND_{order}_EXIT__:'
            shell.stdin.write(command + '\n')
            if submitted:
                shell.stdin.write(submitted + '\n')
            shell.stdin.write("printf '" + marker + "%s\\n' \"$?\"\n")
            shell.stdin.flush()
            lines = []
            while True:
                line = shell.stdout.readline()
                assert line, ('shell ended', command, lines)
                if line.startswith(marker):
                    status = int(line[len(marker):])
                    break
                lines.append(line)
            output = ''.join(lines)
            print(json.dumps(dict(run_id=run_id, order=order, cwd=str(root),
                                  command=command, input_id=submitted, exit=status,
                                  stdout_stderr=output)), flush=True)
            assert status == 0, (command, status, output)
            for line in lines:
                for key in ('target_id', 'evidence_id', 'turn_id'):
                    if line.startswith(key + ': '):
                        ids[key] = line.split(': ', 1)[1].strip()
                if line.startswith('{'):
                    value = json.loads(line).get('data', {}).get('value', {})
                    if 'case cognitive plan' in command:
                        plans.append(value['plan'])
                    elif 'case cognitive realize' in command:
                        realizations.append(value)
                    elif 'case conversation turn show' in command:
                        turns.append(value['turn'])
    assert len(plans) == 3 and len(realizations) == 2
    assert plans[0]['selected_target_id'] != plans[1]['selected_target_id']
    assert plans[0]['execution_lane_id'] != plans[1]['execution_lane_id']
    assert 'trust_not_approved' in plans[1]['arbitration']['candidates'][0]['exclusions']
    assert plans[2]['route'] == 'unresolved' and plans[2].get('selected_target_id') is None
    assert len(turns) == 2 and turns[0] == turns[1]
    assert all(r['provider_execution_performed_now'] for r in realizations)
    print(json.dumps(dict(run_id=run_id, manual_replay='PASS', human_acceptance='not_claimed',
                          proof_class='product', provider_mode='loopback_fixture',
                          commands=order, exact_realizations=2, pinned_refusal=True,
                          immutable_turn=True, cleanup='disposable_home_removed')), flush=True)
finally:
    shell.stdin.close()
    shell.wait(timeout=10)
    for peer in peers:
        peer.terminate()
        try:
            peer.wait(timeout=5)
        except subprocess.TimeoutExpired:
            peer.kill()
            peer.wait(timeout=5)
